cmake_minimum_required(VERSION 3.29)
project(label_smoke LANGUAGES C CXX)
set(LVGL_DIR "" CACHE PATH "LVGL checkout")
set(CONFIG_LV_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
add_subdirectory("${LVGL_DIR}" lvgl)
add_executable(label_smoke main.cpp ../../src/widget/Label.cpp)
target_compile_features(label_smoke PRIVATE cxx_std_17)
target_include_directories(label_smoke PRIVATE ../../include ../../../ui-lvgl/src ../../../framework/src)
target_link_libraries(label_smoke PRIVATE lvgl::lvgl)
if(MSVC)
    target_compile_options(label_smoke PRIVATE /UNDEBUG)
else()
    target_compile_options(label_smoke PRIVATE -UNDEBUG)
endif()
enable_testing()
foreach(case IN ITEMS unchanged pause disable move borrowed parent_first layout)
    add_test(NAME label_${case} COMMAND label_smoke ${case})
endforeach()
